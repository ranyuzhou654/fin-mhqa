### 3.1 `ingest/extract.py`

```python
import fitz  # PyMuPDF
import pdfplumber
from dataclasses import dataclass, field

@dataclass
class RawTable:
    page_no: int
    rows: list[list[str]]
    bbox: tuple                     # 用于找表上方的标题/单位行
    title_hint: str = ""
    unit: str = ""
    currency: str = ""

@dataclass
class Page:
    company: str
    page_no: int
    text: str
    tables: list[RawTable] = field(default_factory=list)

def _norm_cell(c):
    return (c or "").replace("\n", " ").strip()

def extract_pdf(path: str, company: str) -> list[Page]:
    pages = []
    doc = fitz.open(path)
    with pdfplumber.open(path) as pl:
        for i, page in enumerate(doc):
            text = page.get_text("text")
            tables = []
            try:
                plp = pl.pages[i]
                for t in plp.find_tables():
                    rows = [[_norm_cell(c) for c in r] for r in t.extract() if r]
                    rows = [r for r in rows if any(r)]
                    if len(rows) >= 2 and max(len(r) for r in rows) >= 2:
                        tables.append(RawTable(i + 1, rows, t.bbox))
                # 表上方 60pt 内的文本行：找标题与单位
                for tb in tables:
                    x0, top, x1, bottom = tb.bbox
                    above = plp.crop((0, max(top - 60, 0), plp.width, top)).extract_text() or ""
                    tb.title_hint, tb.unit, tb.currency = parse_table_header(above)
            except Exception:
                pass
            pages.append(Page(company, i + 1, text, tables))
    return pages

import re
UNIT_PAT = re.compile(r"单位[:：]\s*(元|千元|万元|百万元|亿元)")
CUR_PAT = re.compile(r"币种[:：]\s*(人民币|美元|港币)")
def parse_table_header(txt: str):
    lines = [l.strip() for l in txt.split("\n") if l.strip()]
    unit = (UNIT_PAT.search(txt) or [None, ""])[1]
    cur = (CUR_PAT.search(txt) or [None, ""])[1]
    title = ""
    for l in reversed(lines):
        if UNIT_PAT.search(l) or CUR_PAT.search(l): continue
        if len(l) < 40 and (l.endswith("表") or re.search(r"情况|构成|明细|数据|指标", l)):
            title = l; break
    return title, unit, cur
```

### 3.2 `ingest/clean.py`

```python
import re

HEADER_PAT = re.compile(r"^.{0,30}2025\s*年半年度报告(全文)?.{0,10}$", re.M)
PAGENO_PAT = re.compile(r"^\s*\d{1,3}\s*(/\s*\d{1,3})?\s*$", re.M)
TOC_LINE = re.compile(r"[\.…]{4,}\s*\d+\s*$")

L1 = re.compile(r"^第[一二三四五六七八九十]+节\s*\S")
L2 = re.compile(r"^[一二三四五六七八九十]+、\s*\S")
L3 = re.compile(r"^（[一二三四五六七八九十]+）\s*\S")
L4 = re.compile(r"^\d{1,2}[\.、]\s*\S")

def heading_level(line: str) -> int:
    if len(line) > 40: return 0
    if L1.match(line): return 1
    if L2.match(line): return 2
    if L3.match(line): return 3
    if L4.match(line) and not re.search(r"\d[,\.]\d{3}", line): return 4   # 排除 "1.23 亿元" 这类
    return 0

def is_toc_page(text: str) -> bool:
    return len(TOC_LINE.findall(text)) >= 8

def clean_text(t: str) -> str:
    t = HEADER_PAT.sub("", t)
    t = PAGENO_PAT.sub("", t)
    t = t.replace("\u3000", " ").replace("\xa0", " ")
    lines = [l.strip() for l in t.split("\n") if l.strip()]
    out = []
    for l in lines:
        if out and heading_level(l) == 0 and heading_level(out[-1]) == 0 \
           and not re.search(r"[。；：！？）)]$", out[-1]):
            out[-1] += l                       # 合并被换行切断的句子
        else:
            out.append(l)
    return "\n".join(out)
```

### 3.3 `ingest/structure.py`（章节树与跨页表合并）

```python
from .clean import clean_text, heading_level, is_toc_page

HEAD_WORDS = ("项目", "本期", "本报告期", "上年同期", "期末", "期初", "金额", "比例")

def looks_like_header(row) -> bool:
    return any(any(w in c for w in HEAD_WORDS) for c in row)

def merge_cross_page_tables(pages):
    """下一页首表与上一页尾表列数相同且首行不像表头 → 拼接"""
    for a, b in zip(pages, pages[1:]):
        if not a.tables or not b.tables: continue
        ta, tb = a.tables[-1], b.tables[0]
        if len(ta.rows[0]) == len(tb.rows[0]) and not looks_like_header(tb.rows[0]):
            ta.rows.extend(tb.rows)
            ta.page_end = tb.page_no
            b.tables.pop(0)
    return pages

def walk_sections(pages):
    """产出 (section_stack, item)；item 为 ('para', text) 或 ('table', RawTable)"""
    stack = []                                   # [(level, title)]
    for p in pages:
        if is_toc_page(p.text): continue
        text = clean_text(p.text)
        for line in text.split("\n"):
            lv = heading_level(line)
            if lv:
                while stack and stack[-1][0] >= lv: stack.pop()
                stack.append((lv, line))
                continue
            yield list(stack), p, ("para", line)
        for tb in p.tables:
            yield list(stack), p, ("table", tb)
```

### 3.4 `ingest/chunk.py`

```python
import json, uuid, re
from simhash import Simhash
from transformers import AutoTokenizer
from .structure import walk_sections, merge_cross_page_tables

tok = AutoTokenizer.from_pretrained("/root/autodl-tmp/models/bge-m3")
ntok = lambda s: len(tok(s, add_special_tokens=False)["input_ids"])

MIN_TOK, TARGET_TOK, MAX_TOK, OVERLAP_TOK = 80, 350, 450, 70

def breadcrumb(company, stack):
    path = " > ".join(t for _, t in stack) if stack else "正文"
    return f"【{company} | 2025年半年度报告 | {path}】", path

def new_chunk(company, stack, page, ctype, body, **meta):
    bc, path = breadcrumb(company, stack)
    return {"chunk_id": uuid.uuid4().hex[:12], "company": company, "report_period": "2025H1",
            "section_path": path, "section_level": stack[-1][0] if stack else 0,
            "page_start": page, "page_end": meta.pop("page_end", page),
            "chunk_type": ctype, "text": bc + "\n" + body, "n_tokens": ntok(body), **meta}

# ---------- 正文 ----------
def split_long_para(text):
    sents = re.split(r"(?<=[。；！？])", text)
    out, buf = [], ""
    for s in sents:
        if ntok(buf + s) > MAX_TOK and ntok(buf) >= 100:
            out.append(buf); buf = s
        else:
            buf += s
    if buf: out.append(buf)
    return out

class TextChunker:
    def __init__(self, company):
        self.company, self.buf, self.stack, self.page = company, [], None, None
        self.out = []
    def _flush(self):
        if not self.buf: return
        body = "\n".join(self.buf)
        self.out.append(new_chunk(self.company, self.stack, self.page, "text", body))
        tail = self.buf[-1] if ntok(self.buf[-1]) <= OVERLAP_TOK else ""
        self.buf = [tail] if tail else []
    def push(self, stack, page, para):
        if self.stack is not None and stack[:3] != self.stack[:3]:   # L3 及以上变化硬切
            self._flush(); self.buf = []
        self.stack, self.page = stack, page.page_no
        for piece in ([para] if ntok(para) <= MAX_TOK else split_long_para(para)):
            self.buf.append(piece)
            if ntok("\n".join(self.buf)) >= TARGET_TOK: self._flush()
    def finish(self):
        self._flush(); self.buf = []
        return merge_short(self.out)

def merge_short(chunks):
    out = []
    for c in chunks:
        if out and c["n_tokens"] < MIN_TOK and out[-1]["section_path"] == c["section_path"]:
            out[-1]["text"] += "\n" + c["text"].split("\n", 1)[1]
            out[-1]["n_tokens"] += c["n_tokens"]; out[-1]["page_end"] = c["page_end"]
        else:
            out.append(c)
    return out

# ---------- 表格 ----------
def fill_merged_cells(rows):
    w = max(len(r) for r in rows)
    rows = [r + [""] * (w - len(r)) for r in rows]
    for i, r in enumerate(rows):
        for j, c in enumerate(r):
            if not c and i > 0 and j == 0: r[j] = rows[i-1][0]       # 首列向下填充
            elif not c and j > 0 and i == 0: r[j] = r[j-1]           # 表头向右填充
    return rows

def to_md(rows):
    head = "| " + " | ".join(rows[0]) + " |"
    sep = "|" + "---|" * len(rows[0])
    body = "\n".join("| " + " | ".join(r) + " |" for r in rows[1:])
    return f"{head}\n{sep}\n{body}"

def split_table(rows, max_rows=20, max_cols=12):
    header, body = rows[0], rows[1:]
    col_groups = [list(range(len(header)))]
    if len(header) > max_cols:                                        # 宽表按列分组，保留首列
        col_groups = [[0] + list(range(s, min(s + max_cols - 1, len(header))))
                      for s in range(1, len(header), max_cols - 1)]
    for cols in col_groups:
        for s in range(0, len(body), max_rows):
            yield [[header[j] for j in cols]] + [[r[j] for j in cols] for r in body[s:s + max_rows]]

def kv_sentences(company, title, rows, unit):
    header = rows[0]
    u = f"（单位：{unit}）" if unit else ""
    for r in rows[1:]:
        name = r[0]
        if not name or not any(r[1:]): continue
        parts = [f"{header[j]} {r[j]}" for j in range(1, len(r)) if r[j] and header[j]]
        if parts:
            yield f"【{company} | {title}】{name} " + "，".join(parts) + u

def table_chunks(company, stack, tb):
    rows = fill_merged_cells(tb.rows)
    title = tb.title_hint or (stack[-1][1] if stack else "表格")
    parent = uuid.uuid4().hex[:12]
    head = f"表：{title}" + (f"　单位：{tb.unit}" if tb.unit else "") + (f"　币种：{tb.currency}" if tb.currency else "")
    for k, part in enumerate(split_table(rows)):
        md = new_chunk(company, stack, tb.page_no, "table_md", head + "\n" + to_md(part),
                       table_title=title, unit=tb.unit, currency=tb.currency,
                       parent_table_id=parent, part=k, page_end=getattr(tb, "page_end", tb.page_no))
        yield md
        kv = "\n".join(kv_sentences(company, title, part, tb.unit))
        if kv:
            yield new_chunk(company, stack, tb.page_no, "table_kv", kv,
                            table_title=title, unit=tb.unit, currency=tb.currency,
                            parent_table_id=parent, part=k)

# ---------- 主流程 ----------
def dedup(chunks, dist=3):
    seen, out = [], []
    for c in sorted(chunks, key=lambda c: -c["section_level"]):     # 深层优先保留
        body = c["text"].split("\n", 1)[1]
        h = Simhash(body)
        if any(h.distance(s) <= dist for s in seen): continue
        seen.append(h); out.append(c)
    return out

def chunk_pages(company, pages):
    pages = merge_cross_page_tables(pages)
    tc, tbl = TextChunker(company), []
    for stack, page, (kind, item) in walk_sections(pages):
        if kind == "para":
            tc.push(stack, page, item)
        else:
            tbl.extend(table_chunks(company, stack, item))
    return dedup(tc.finish() + tbl)

def run(pdf_dir, out_path, company_map):

    with open(out_path, "w") as f:
        for fname, company in company_map.items():
            for c in chunk_pages(company, extract_pdf(f"{pdf_dir}/{fname}", company)):
                f.write(json.dumps(c, ensure_ascii=False) + "\n")
```

### 4.2 `retrieval/retriever.py`

```python
import json, pickle, torch, numpy as np, faiss, bm25s, jieba
from FlagEmbedding import BGEM3FlagModel, FlagReranker

class HybridRetriever:
    def __init__(self, index_dir, model_dir):
        self.meta = [json.loads(l) for l in open(f"{index_dir}/chunk_meta.jsonl")]
        self.faiss = faiss.read_index(f"{index_dir}/faiss.index")
        self.bm25 = pickle.load(open(f"{index_dir}/bm25.pkl", "rb"))
        gpu = torch.cuda.is_available()          # CPU 实例与 GPU 实例共用同一份代码
        self.embed = BGEM3FlagModel(f"{model_dir}/bge-m3", use_fp16=gpu)
        self.reranker = FlagReranker(f"{model_dir}/bge-reranker-v2-m3", use_fp16=gpu)
        self.id2chunk = {c["chunk_id"]: i for i, c in enumerate(self.meta)}
        self.parent_md = {}                        # parent_table_id -> [idx of table_md parts]
        for i, c in enumerate(self.meta):
            if c["chunk_type"] == "table_md":
                self.parent_md.setdefault(c["parent_table_id"], []).append(i)

    def dense(self, q, k=30, company=None):
        v = self.embed.encode([q])["dense_vecs"].astype("float32")
        faiss.normalize_L2(v)
        scores, ids = self.faiss.search(v, k * 3 if company else k)
        res = [(int(i), float(s)) for i, s in zip(ids[0], scores[0]) if i >= 0]
        return self._filter(res, company)[:k]

    def sparse(self, q, k=30, company=None):
        toks = [list(jieba.cut(q))]
        ids, scores = self.bm25.retrieve(toks, k=k * 3 if company else k)
        res = [(int(i), float(s)) for i, s in zip(ids[0], scores[0])]
        return self._filter(res, company)[:k]

    def _filter(self, res, company):
        if not company: return res
        return [(i, s) for i, s in res if self.meta[i]["company"] == company]

    @staticmethod
    def rrf(lists, k=60):
        agg = {}
        for lst in lists:
            for rank, (i, _) in enumerate(lst):
                agg[i] = agg.get(i, 0.0) + 1.0 / (k + rank + 1)
        return sorted(agg.items(), key=lambda x: -x[1])

    def rerank(self, q, cand_ids, top_k=5):
        pairs = [[q, self.meta[i]["text"]] for i in cand_ids]
        scores = self.reranker.compute_score(pairs, normalize=True)
        order = np.argsort(scores)[::-1][:top_k]
        return [(cand_ids[j], float(scores[j])) for j in order]

    def _fmt(self, i, s, attached=False):
        m = self.meta[i]
        return {"chunk_id": m["chunk_id"], "company": m["company"], "section": m["section_path"],
                "chunk_type": m["chunk_type"], "score": round(s, 3), "attached": attached,
                "text": m["text"][:1500]}

    def search(self, q, mode="hybrid", top_k=5, company=None, expand_parent=True, n_cand=30):
        if mode == "dense":
            cand = self.dense(q, n_cand, company)
        elif mode == "bm25":
            cand = self.sparse(q, n_cand, company)
        else:
            cand = self.rrf([self.dense(q, n_cand, company), self.sparse(q, n_cand, company)])[:n_cand]
        ranked = self.rerank(q, [i for i, _ in cand], top_k)
        out = [self._fmt(i, s) for i, s in ranked]
        if expand_parent:                          # 命中键值句时附带其完整 Markdown 表格块
            have = {r["chunk_id"] for r in out}
            for i, s in ranked:
                m = self.meta[i]
                if m["chunk_type"] == "table_kv":
                    for j in self.parent_md.get(m["parent_table_id"], []):
                        if self.meta[j].get("part") == m.get("part") and self.meta[j]["chunk_id"] not in have:
                            out.append(self._fmt(j, s, attached=True)); have.add(self.meta[j]["chunk_id"])
        return out
```

### 5.4 `agent/llm.py`

```python
import json, re, os, yaml, asyncio
from openai import OpenAI
from tenacity import retry, stop_after_attempt, wait_exponential

CFG = yaml.safe_load(open("configs/llm.yaml"))["endpoints"]

class LLM:
    def __init__(self, role="teacher"):
        c = CFG[role]
        self.client = OpenAI(base_url=c["base_url"], api_key=os.environ.get(c["api_key_env"], "EMPTY"))
        self.model = c["model"]

    @retry(stop=stop_after_attempt(4), wait=wait_exponential(min=1, max=20))
    def chat(self, prompt, temperature=0.3, max_tokens=2048, system=None):
        msgs = ([{"role": "system", "content": system}] if system else []) + [{"role": "user", "content": prompt}]
        r = self.client.chat.completions.create(model=self.model, temperature=temperature,
                                                max_tokens=max_tokens, messages=msgs)
        return r.choices[0].message.content or ""

    def chat_json(self, prompt, fallback=None, **kw):
        for attempt in range(2):
            txt = self.chat(prompt, **{**kw, "temperature": kw.get("temperature", 0.3) + 0.4 * attempt})
            txt = re.sub(r"<think>.*?</think>", "", txt, flags=re.S)
            m = re.search(r"\{.*\}", txt, flags=re.S)
            if m:
                try: return json.loads(m.group(0))
                except json.JSONDecodeError: pass
        return fallback
```

### 5.5 `agent/nodes.py`

```python
import json, re
from .llm import LLM
from .tools import call_tool, format_results, infer_company, ALT
from . import prompts as P

MAX_REPLAN, MAX_TOOL_CALLS, MAX_SUBQ = 2, 12, 4
llm = LLM("teacher")           # 由 graph.build_graph 注入，可换 local_32b

def _fill(text, plan):
    for q in plan:
        if q["status"] == "resolved":
            text = text.replace("{{" + q["id"] + "}}", q["conclusion"])
    return text

# ---------- Planner ----------
def planner(state):
    fb = state.get("feedback")
    fb_block = P.FEEDBACK_BLOCK.format(**fb, tried=state["tried"]) if fb else ""
    out = llm.chat_json(P.PLANNER.format(question=state["question"], feedback_block=fb_block),
                        fallback={"plan": [{"id": "Q1", "question": state["question"],
                                            "depends_on": [], "tool": "search_hybrid", "company": None}]})
    new = []
    resolved_old = {q["id"]: q for q in state["plan"] if q["status"] == "resolved"}
    for q in out["plan"][:MAX_SUBQ]:
        if q["id"] in resolved_old:                       # Replan 时保留已解决子问题
            new.append(resolved_old[q["id"]]); continue
        new.append({**q, "status": "pending", "conclusion": "", "evidence_ids": [], "confidence": 0.0})
    return {"plan": new, "current": 0, "feedback": None,
            "trajectory": [{"role": "planner", "replan": state["replan_count"], "input": fb, "output": out}]}

# ---------- Executor ----------
def _retrieve(state, sq, tool, query, company):
    results, err = call_tool(tool, {"query": query, "company": company})
    return results, err

def executor(state):
    i = state["current"]; sq = dict(state["plan"][i])
    plan = list(state["plan"])
    if sq["status"] == "resolved":                        # Replan 保留下来的，跳过
        return {"current": i + 1}
    known = {q["id"]: q["conclusion"] for q in plan[:i] if q["status"] == "resolved"}
    if any(plan[[q["id"] for q in plan].index(d)]["status"] != "resolved" for d in sq.get("depends_on", [])):
        sq["status"] = "blocked"; plan[i] = sq
        return {"plan": plan, "current": i + 1,
                "trajectory": [{"role": "executor", "subq_id": sq["id"], "blocked": True}]}

    query = _fill(sq["question"], plan)
    if sq["tool"] == "none":                              # 纯计算
        out = llm.chat_json(P.CALC.format(subq=query, known=known),
                            fallback={"conclusion": "INSUFFICIENT", "evidence_ids": [], "confidence": 0, "missing": "calc failed"})
        sq.update(conclusion=out["conclusion"], evidence_ids=[], confidence=out["confidence"],
                  status="failed" if out["conclusion"] == "INSUFFICIENT" else "resolved")
        plan[i] = sq
        return {"plan": plan, "current": i + 1, "trajectory": [{"role": "executor", "subq_id": sq["id"], "calc": out}]}

    company = sq.get("company") or infer_company(query)
    tool = sq["tool"]
    results, err = _retrieve(state, sq, tool, query, company)
    pool = {r["chunk_id"]: r for r in results}
    tried = [(tool, query, company)]
    calls = 1
    out = _conclude(query, known, results)
    retry = False
    if (out["conclusion"] == "INSUFFICIENT" or out["confidence"] < 0.5) and state["tool_calls"] + calls < MAX_TOOL_CALLS:
        retry = True                                      # 局部重试：换一路工具
        alt = ALT[tool]
        r2, err2 = _retrieve(state, sq, alt, query, company)
        pool.update({r["chunk_id"]: r for r in r2}); tried.append((alt, query, company)); calls += 1
        out = _conclude(query, known, list(pool.values()))
    valid_ids = [c for c in out.get("evidence_ids", []) if c in pool]
    sq.update(conclusion=out["conclusion"], evidence_ids=valid_ids, confidence=out.get("confidence", 0),
              status="failed" if out["conclusion"] == "INSUFFICIENT" or not valid_ids else "resolved")
    plan[i] = sq
    return {"plan": plan, "current": i + 1, "evidence_pool": pool, "tried": tried,
            "tool_calls": state["tool_calls"] + calls,
            "trajectory": [{"role": "executor", "subq_id": sq["id"], "tool": tool, "query": query,
                            "company": company, "retry": retry, "error": err,
                            "tool_results": list(pool.values()), "output": out}]}

def _conclude(query, known, results):
    if not results:
        return {"conclusion": "INSUFFICIENT", "evidence_ids": [], "confidence": 0, "missing": "no results"}
    return llm.chat_json(P.EXECUTOR.format(subq=query, known=known, evidence=format_results(results)),
                         fallback={"conclusion": "INSUFFICIENT", "evidence_ids": [], "confidence": 0, "missing": "parse failed"})

# ---------- Verifier ----------
NUM = re.compile(r"\d[\d,]*\.?\d*")
def _precheck(plan, pool):
    """结论中的数字是否出现在引用证据里（去掉千分位后比对）"""
    rep = {}
    for q in plan:
        if q["status"] != "resolved" or q["tool"] == "none": continue
        ev = " ".join(pool.get(c, {}).get("text", "") for c in q["evidence_ids"]).replace(",", "")
        nums = [n.replace(",", "") for n in NUM.findall(q["conclusion"]) if len(n) >= 3]
        rep[q["id"]] = all(n in ev for n in nums) if nums else True
    return rep

def verifier(state):
    pre = _precheck(state["plan"], state["evidence_pool"])
    plan_view = [{k: q[k] for k in ("id", "question", "status", "conclusion", "evidence_ids")} for q in state["plan"]]
    out = llm.chat_json(P.VERIFIER.format(question=state["question"],
                                          plan=json.dumps(plan_view, ensure_ascii=False, indent=1), precheck=pre),
                        fallback={"sufficient": False, "resolved_subquestions": [], "unresolved_subquestions": ["all"],
                                  "failure_reason": "verifier parse failed", "suggestion": "换用 search_hybrid 并加入期间词"})
    if any(q["status"] != "resolved" for q in state["plan"]):
        out["sufficient"] = False                          # 代码硬约束：有未解决子问题不得通过
    return {"feedback": None if out["sufficient"] else out,
            "trajectory": [{"role": "verifier", "precheck": pre, "output": out}]}

# ---------- Synthesizer ----------
CITE = re.compile(r"引用[:：]\s*\[(.*?)\]")
def synthesizer(state):
    ids = {c for q in state["plan"] for c in q["evidence_ids"]}
    ev = [state["evidence_pool"][c] for c in ids if c in state["evidence_pool"]]
    plan_view = [{k: q[k] for k in ("id", "question", "conclusion", "evidence_ids")} for q in state["plan"]]
    text = llm.chat(P.SYNTHESIZER.format(question=state["question"],
                                         plan=json.dumps(plan_view, ensure_ascii=False, indent=1),
                                         evidence=format_results(ev)), temperature=0.1)
    m = CITE.search(text)
    cited = [c.strip() for c in m.group(1).split(",") if c.strip()] if m else []
    cited = [c for c in cited if c in state["evidence_pool"]]
    return {"final_answer": text, "cited_ids": cited,
            "trajectory": [{"role": "synthesizer", "output": text, "cited": cited}]}

# ---------- 路由 ----------
def after_executor(state):
    if state["current"] < len(state["plan"]) and state["tool_calls"] < MAX_TOOL_CALLS:
        return "executor"
    return "verifier"

def after_verifier(state):
    if state["feedback"] is None: return "synthesizer"
    if state["replan_count"] >= MAX_REPLAN or state["tool_calls"] >= MAX_TOOL_CALLS: return "synthesizer"
    return "replan"

def bump_replan(state):
    return {"replan_count": state["replan_count"] + 1}
```

### 5.6 `agent/graph.py`

```python
from langgraph.graph import StateGraph, END
from langgraph.checkpoint.sqlite import SqliteSaver
from .state import AgentState
from . import nodes as N
from .llm import LLM

def build_graph(llm_role="teacher", checkpoint_path=None):
    N.llm = LLM(llm_role)
    g = StateGraph(AgentState)
    for name, fn in [("planner", N.planner), ("executor", N.executor), ("verifier", N.verifier),
                     ("replan", N.bump_replan), ("synthesizer", N.synthesizer)]:
        g.add_node(name, fn)
    g.set_entry_point("planner")
    g.add_edge("planner", "executor")
    g.add_conditional_edges("executor", N.after_executor, {"executor": "executor", "verifier": "verifier"})
    g.add_conditional_edges("verifier", N.after_verifier, {"replan": "replan", "synthesizer": "synthesizer"})
    g.add_edge("replan", "planner")
    g.add_edge("synthesizer", END)
    cp = SqliteSaver.from_conn_string(checkpoint_path) if checkpoint_path else None
    return g.compile(checkpointer=cp)

def init_state(question):
    return {"question": question, "plan": [], "current": 0, "evidence_pool": {}, "tried": [],
            "feedback": None, "replan_count": 0, "tool_calls": 0,
            "final_answer": "", "cited_ids": [], "trajectory": []}

def run(question, graph, thread_id="0"):
    return graph.invoke(init_state(question), config={"configurable": {"thread_id": thread_id, "recursion_limit": 60}})
```

### 5.7 批量运行器 `agent/batch.py`

数据合成和评测都要对上千题跑图，统一用线程池 + 断点续跑：

```python
import json, os, concurrent.futures as cf
from .graph import build_graph, run

def run_batch(items, out_path, llm_role="teacher", workers=32, temperature_note=None):
    done = {json.loads(l)["id"] for l in open(out_path)} if os.path.exists(out_path) else set()
    todo = [it for it in items if it["id"] not in done]
    graph = build_graph(llm_role)
    with open(out_path, "a") as f, cf.ThreadPoolExecutor(workers) as ex:
        futs = {ex.submit(run, it["question"], graph, it["id"]): it for it in todo}
        for fut in cf.as_completed(futs):
            it = futs[fut]
            try:
                st = fut.result()
                rec = {"id": it["id"], "question": it["question"], "final_answer": st["final_answer"],
                       "cited_ids": st["cited_ids"], "plan": st["plan"], "trajectory": st["trajectory"],
                       "tool_calls": st["tool_calls"], "replan_count": st["replan_count"]}
            except Exception as e:
                rec = {"id": it["id"], "error": repr(e)}
            f.write(json.dumps(rec, ensure_ascii=False) + "\n"); f.flush()
```

`workers` 不超过 `configs/llm.yaml` 中 `max_concurrency`；LangGraph 图对象线程安全（无共享可变状态），`N.llm` 的 OpenAI client 也是线程安全的。

### 5.8 扁平模式运行器 `agent/flat.py`（4B 推理用）

4B 训练后以单模型多轮 tool_call 方式运行，不走四角色图：

```python
import json, re
from openai import OpenAI
from .tools import TOOLS, call_tool, format_results

SYSTEM = "你是金融财报多跳问答助手。通过调用检索工具逐步获取证据，最后一行以『答案：…；引用：[chunk_id]』作答。"

def run_flat(question, base_url="http://127.0.0.1:8001/v1", model="qwen3-4b", max_turns=8):
    client = OpenAI(base_url=base_url, api_key="EMPTY")
    msgs = [{"role": "system", "content": SYSTEM}, {"role": "user", "content": question}]
    traj, pool = [], {}
    for turn in range(max_turns):
        r = client.chat.completions.create(model=model, messages=msgs, tools=TOOLS, temperature=0.3, max_tokens=1024)
        m = r.choices[0].message
        msgs.append(m.model_dump(exclude_none=True))
        if not m.tool_calls:
            break
        for tc in m.tool_calls:
            args = json.loads(tc.function.arguments)
            results, err = call_tool(tc.function.name, args)
            pool.update({x["chunk_id"]: x for x in results})
            traj.append({"role": "executor", "tool": tc.function.name, "query": args.get("query"),
                         "tool_results": results, "error": err})
            msgs.append({"role": "tool", "tool_call_id": tc.id, "content": format_results(results) or "无结果"})
    final = msgs[-1].get("content", "") if msgs[-1]["role"] == "assistant" else ""
    mm = re.search(r"引用[:：]\s*\[(.*?)\]", final)
    cited = [c.strip() for c in mm.group(1).split(",") if c.strip() in pool] if mm else []
    return {"final_answer": final, "cited_ids": cited, "trajectory": traj, "evidence_pool": pool}
```

评测脚本对 32B 调 `graph.run`，对 4B 调 `flat.run_flat`，两者返回同一结构。

```python
    if len(hops) < 2: return False
    if len({h["chunk_id"] for h in hops}) < len(hops): return False       # 证据不能重复
```

### 7.1 `eval/metrics.py`

```python
from finmhqa.utils.num import parse_number, unit_normalize

def correctness(pred, gold, answer_type, judge):
    if answer_type == "number":
        p, g = parse_number(pred), parse_number(gold)
        if p is not None and g is not None:
            return float(abs(unit_normalize(p) - unit_normalize(g)) <= 0.01 * abs(unit_normalize(g)))
    return judge.correct(pred, gold)          # 文本题走 judge，返回 0/1

def faithfulness(answer, cited_chunks, judge):
    claims = judge.extract_claims(answer)     # list[str]
    if not claims: return 0.0
    supported = sum(judge.supported(c, cited_chunks) for c in claims)
    return supported / len(claims)

def context_precision(cited_ids, gold_ids):
    cited = set(cited_ids)
    return len(cited & set(gold_ids)) / max(len(cited), 1)

def hop_evidence_hit(trajectory, hops):
    hits = []
    for h in hops:
        seen = {cid for step in trajectory if step["role"] == "executor" for cid in step["results"]}
        hits.append(h["chunk_id"] in seen)
    return sum(hits) / len(hits)
```

### 9.3 `configs/tools_verl.yaml` 与工具类

verl 的多轮工具需要实现一个 `BaseTool` 子类：

```python
# src/finmhqa/train/verl_tools.py
from verl.tools.base_tool import BaseTool
from verl.tools.schemas import OpenAIFunctionToolSchema, ToolResponse
import httpx, uuid

class RetrieveTool(BaseTool):
    MODE = "hybrid"
    def __init__(self, config, tool_schema):
        super().__init__(config, tool_schema)
        self.url = config.get("url", "http://127.0.0.1:8100/search")
        self._inst = {}

    async def create(self, instance_id=None, **kw):
        iid = instance_id or str(uuid.uuid4())
        self._inst[iid] = {"retrieved": []}
        return iid, ToolResponse()

    async def execute(self, instance_id, parameters, **kw):
        try:
            async with httpx.AsyncClient(timeout=30) as c:
                r = await c.post(self.url, json={"query": parameters["query"], "mode": self.MODE,
                                                 "top_k": 5, "company": parameters.get("company")})
            res = r.json()["results"]
        except Exception as e:
            return ToolResponse(text=f"检索失败: {e}"), -0.05, {}
        self._inst[instance_id]["retrieved"].append([x["chunk_id"] for x in res])
        text = "\n\n".join(f"[{x['chunk_id']}] ({x['company']} | {x['section']})\n{x['text']}" for x in res)
        return ToolResponse(text=text), 0.0, {"retrieved": [x["chunk_id"] for x in res]}

    async def calc_reward(self, instance_id, **kw):
        return 0.0          # 主 reward 在 reward.py 统一算

    async def release(self, instance_id, **kw):
        self._inst.pop(instance_id, None)

class BM25Tool(RetrieveTool):   MODE = "bm25"
class DenseTool(RetrieveTool):  MODE = "dense"
```

### 9.4 `train/reward.py`（自定义 reward，两阶段共用，权重由环境变量控制）

```python
import json, os, re
from finmhqa.utils.num import parse_number, unit_normalize

W = dict(c=float(os.getenv("RW_C", 1.0)), f=float(os.getenv("RW_F", 0.3)),
         p=float(os.getenv("RW_P", 0.3)), h=float(os.getenv("RW_H", 0.8)))
HARD_FAITH = os.getenv("RW_HARD_FAITH", "0") == "1"     # 阶段二打开

ANS_PAT = re.compile(r"答案[:：](../../../.+?)[;；]\s*引用[:：]\s*\[(.*?)\]", re.S)

def compute_score(data_source, solution_str, ground_truth, extra_info=None, **kw):
    gt = json.loads(ground_truth)
    gold = set(gt["gold_chunks"])
    m = ANS_PAT.search(solution_str)
    if not m:
        return -1.0                                           # 格式非法
    pred, cited = m.group(1).strip(), [c.strip() for c in m.group(2).split(",") if c.strip()]
    r_format = 0.1

    # Correctness
    if gt["answer_type"] == "number":
        p, g = parse_number(pred), parse_number(gt["answer"])
        r_c = float(p is not None and g is not None and abs(unit_normalize(p) - unit_normalize(g)) <= 0.01 * abs(unit_normalize(g)))
    else:
        r_c = float(gt["answer"] in pred or pred in gt["answer"])

    # Faithfulness（规则版）：答案关键数值/实体是否出现在引用 chunk 文本中
    cited_text = " ".join(CHUNK_TEXT.get(c, "") for c in cited)
    key = gt["answer"] if gt["answer_type"] != "number" else str(parse_number(gt["answer"]))
    r_f = float(bool(cited) and key_in_text(key, cited_text))
    if HARD_FAITH and r_f == 0.0:
        r_c = 0.0                                             # 阶段二硬约束：不被证据支持的正确答案不计分

    # Context precision
    r_p = len(set(cited) & gold) / max(len(set(cited)), 1)

    # 分跳证据匹配：从 extra_info 的工具轨迹里取每轮检索到的 chunk
    retrieved = set()
    for step in (extra_info or {}).get("tool_metrics", []):
        retrieved.update(step.get("retrieved", []))
    r_h = sum(g in retrieved for g in gold) / len(gold)

    return W["c"] * r_c + W["f"] * r_f + W["p"] * r_p + W["h"] * r_h + r_format
```

> `extra_info` 中工具轨迹的字段名取决于 verl 版本如何把 `execute` 返回的 metrics 透传给 reward 函数；若不透传，改为在 `RetrieveTool.calc_reward` 中计算 `r_h` 并由 verl 汇总（`tool_reward`）。两条路都行，先在小 batch 上打印确认。

## 11. 分阶段 AutoDL 操作流程

```
[阶段 A] M5 CPU 实例 (常开 1–2 周，M1–M4 全在这里)
  autodl_setup.sh（CPU 版 torch）→ 下 bge-m3 / reranker → M1 抽取+chunk → inspect_chunks 验收
  → M2 build_index（CPU embedding 约 30–60 min）→ 起检索服务（screen）→ 30 条查询验收
  → M3 用 API teacher 调试图 → dev20 验收
  → M4 seed_qa → expand → rule_filter → llm_check → reachability（run_batch, workers=32, 每题 2 次）
  → judge（不同模型）→ split → 写 TEST_LOCK
  → traj_to_sharegpt / to_verl_parquet 也在这里做完
  → 关机（数据盘保留 index/ data/ ）

[阶段 B] 2× 4090 (约 1 天)
  GPU 环境 → 起检索服务（GPU 版，同一 index/）→ serve_32b.sh
  → M5 跑 32B 固定 Pipeline 基线（3 runs，judge 走 API）→ 冻结 results/BASELINE.md → 关机

[阶段 C] 1× 4090 (半天)
  llamafactory 环境 → M6 SFT → export → serve_4b.sh → flat 模式评测 → 关机

[阶段 D] 2× A800 (1–2 周)
  verl 环境 → 检索服务放 GPU0 → 自检清单 → GRPO S1 → merge → GRPO S2 → merge
  每 20 步 dev 评测曲线推 SwanLab → 关机

[阶段 E] 1× 4090 (1–2 天)
  M8 全部配置评测 + 消融 → REPORT.md
```

每个阶段结束务必：`ckpt/`、`data/`、`index/`、`results/` 都在数据盘；关机前 `rsync` 一份关键结果到本地或对象存储。CPU 实例在阶段 A 结束后也要关机。

### 11.1 CPU 实例上的检索服务调优

- `torch.set_num_threads(os.cpu_count())`；bge-m3 与 reranker 均以 fp32 加载（CPU 无 fp16 加速）
- 服务端做请求级 batch：收到请求后攒 50 ms，把多个 query 的 rerank pair 合并成一次前向
- rerank 候选从 30 降到 20 可将延迟减半，Top-5 质量基本不变
- 数据合成阶段并发 32 时，实测每秒 5–10 次检索可稳定跟上；若 `httpx` 超时增多，先降 `workers` 再考虑换 GPU

## 12. 常见问题速查

| 现象 | 原因 | 处理 |
|---|---|---|
| 4B 输出 tool_call 但 vLLM 不解析 | 未开 `--enable-auto-tool-choice --tool-call-parser hermes` | 补参数 |
| verl rollout 卡住不动 | 工具 HTTP 超时无返回 | 工具端 `timeout=30` + 异常返回文本；检索服务加 worker |
| GRPO reward/std ≈ 0 | 信号稀疏 | 提高 `RW_H`，temperature 0.8→1.0，n 8→16 |
| Faithfulness 下降、Correctness 上升 | 投机作答 | 提前进入阶段二 / 打开 HARD_FAITH |
| 训练后模型只用 search_hybrid | 探索塌缩 | 降 kl_coef，加 entropy_coeff 0.001 |
| test 指标与 dev 指标背离 | dev 泄露或 reward hacking | 检查 dev/test 无交集；看引用 chunk 是否重复刷 precision |
| AutoDL 下载 HF 模型极慢 | 网络 | `HF_ENDPOINT=https://hf-mirror.com` + 学术加速 |
